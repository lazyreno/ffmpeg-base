cmake_minimum_required(VERSION 3.21)
project(zimg LANGUAGES CXX)
file(GLOB_RECURSE ZIMG_SOURCES CONFIGURE_DEPENDS "src/zimg/*.cpp")
list(FILTER ZIMG_SOURCES EXCLUDE REGEX "/(arm|x86)/")
add_library(zimg ${ZIMG_SOURCES})
target_include_directories(zimg PRIVATE src/zimg)
target_compile_features(zimg PRIVATE cxx_std_11)
set_target_properties(zimg PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
install(TARGETS zimg RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES src/zimg/api/zimg.h src/zimg/api/zimg++.hpp DESTINATION include)
