cmake_minimum_required(VERSION 3.21)
project(libgsm LANGUAGES C)
file(GLOB GSM_SOURCES CONFIGURE_DEPENDS "src/*.c")
list(FILTER GSM_SOURCES EXCLUDE REGEX "/toast[^/]*\\.c$")
add_library(gsm ${GSM_SOURCES})
target_include_directories(gsm PRIVATE inc)
set_target_properties(gsm PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON OUTPUT_NAME gsm)
install(TARGETS gsm RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES inc/gsm.h DESTINATION include/gsm)
